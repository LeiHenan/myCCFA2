from transformers.generation.utils import _speculative_sampling

    def test_speculative_sampling(self):
        # assume vocab size 10, input length 5 + 3 generated candidates
        candidate_input_ids = torch.tensor([[8, 0, 3, 9, 8, 1, 4, 5]])  # input tokens
        candidate_logits = torch.tensor([[
            [-10., 10., -10., -10., -10., -10., -10., -10., -10., -10.],  # generated 1

import torch
from transformers import AutoTokenizer, AutoModelForCausalLM, set_seed

def top_k(a, k):
    sorted, indices = a.sort(descending=True)
    return sorted[:k], indices[:k]
def get_dist(outputs, num_samples=10**6, norm_factor=1e7):
    probs = outputs.scores[0][0].softmax(dim=-1)
    print("probs:", top_k(probs, k=5))
    next_std_token = torch.multinomial(probs, num_samples=num_samples, replacement=True)
    counts = torch.bincount(next_std_token, minlength=probs.shape[0]).float()
    print("counts:", top_k(counts, k=5))
    norm_counts = counts.float() / norm_factor
    print("softmaxed_counts:", top_k(counts.softmax(dim=-1), k=5)[0])
    print("norm_counts:", top_k(norm_counts, k=5))
    softmaxed_norm_counts = top_k(norm_counts.softmax(dim=-1), k=5)[0]
    print("softmaxed_norm_counts:", softmaxed_norm_counts)
    return softmaxed_norm_counts

checkpoint = "EleutherAI/pythia-1.4b-deduped"
tokenizer = AutoTokenizer.from_pretrained(checkpoint)
model = AutoModelForCausalLM.from_pretrained(checkpoint)
assistant_model = AutoModelForCausalLM.from_pretrained("EleutherAI/pythia-160m-deduped")
gen_kwargs = dict(**tokenizer("In other words, we", return_tensors="pt"),
                  pad_token_id=tokenizer.eos_token_id,
                  max_new_tokens=1,
                  do_sample=True,
                  temperature=0.1,
                  return_dict_in_generate=True,
                  output_scores=True)
print("Standard sampling:")
set_seed(0)
std_dist = get_dist(model.generate(**gen_kwargs))
print("\nSpeculative decoding:")
set_seed(0)
sd_dist = get_dist(model.generate(**gen_kwargs, assistant_model=assistant_model))
print(std_dist == sd_dist)

In short, I get the token probs for some input for both decoding methods, then I sample 1M times from this prob, and get the token counts (I always look at top 5 scores in all the following computations). If we apply softmax to the counts, they produce an insignificant result of [1., 0., 0., 0., 0.] (tested other inputs too). So I normalize the counts and then apply softmax, and then the output distributions look similar, however, they are still not equal to each other (final output).
