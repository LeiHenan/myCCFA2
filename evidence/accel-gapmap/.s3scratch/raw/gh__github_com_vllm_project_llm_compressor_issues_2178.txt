# ============================================================
# GLM-4.5-Air NVFP4 Quantization Script
# Minimal VRAM · Stable Final Version (FX / SDPA Safe)
# ============================================================

import os
import torch
from datasets import load_dataset
from transformers import AutoTokenizer, AutoModelForCausalLM

from llmcompressor import oneshot
from llmcompressor.modifiers.quantization import QuantizationModifier

# ============================================================
# 0. Global Configuration
# ============================================================

MODEL_ID = "/datas/model_zoo/GLM-4.5-Air"
SAVE_DIR = "/datas/model_zoo/MYmodel/NEW1-GLM-4.5-Air-NVFP4"

MAX_SEQ_LEN = 2048
NUM_CALIBRATION_SAMPLES = 20

torch.set_default_dtype(torch.bfloat16)
torch.backends.cuda.matmul.allow_tf32 = False
torch.backends.cudnn.allow_tf32 = False

# ============================================================
# 1. Load Model (CPU Resident)
# ============================================================

print("Loading model on CPU...")

model = AutoModelForCausalLM.from_pretrained(
    MODEL_ID,
    torch_dtype=torch.bfloat16,
    device_map="cpu",
    trust_remote_code=True,
)

model.eval()
model.requires_grad_(False)

tokenizer = AutoTokenizer.from_pretrained(
    MODEL_ID,
    trust_remote_code=True,
)

print("Model loaded on CPU.\n")

# ============================================================
# 2. Load Calibration Data
# ============================================================

DATASET_DIR = "/root/data/marundong/data/ultrachat_200k/data"

dataset = load_dataset(
    "parquet",
    data_files=f"{DATASET_DIR}/train_sft-00000-of-00003-a3ecf92756993583.parquet",
    split=f"train[:{NUM_CALIBRATION_SAMPLES}]",
)

def messages_to_text(example):
    text = ""
    for msg in example["messages"]:
        text += f'{msg["role"]}: {msg["content"]}\n'
    return {"text": text}

dataset = dataset.map(
    messages_to_text,
    remove_columns=dataset.column_names,
)

# ⚠️ Note: Do not store Tensors inside datasets, only store lists
def tokenize(example):
    out = tokenizer(
        example["text"],
        max_length=MAX_SEQ_LEN,
        truncation=True,
        padding=False,
        add_special_tokens=False,
    )
    return {
        "input_ids": out["input_ids"],          # list[int]
        "attention_mask": out["attention_mask"] # list[int]
    }

dataset = dataset.map(tokenize, remove_columns=["text"])

print("Calibration samples:", len(dataset))

# ============================================================
# Key: FX / SDPA Safe data_collator
# ============================================================

def data_collator(batch):
    # Strict single-sample to control VRAM
    assert len(batch) == 1
    sample = batch[0]

    # ⚠️ Must add batch dimension, otherwise create_causal_mask will fail
    input_ids = torch.tensor(
        sample["input_ids"],
        dtype=torch.long,
        device="cpu",
    ).unsqueeze(0)              # [1, seq_len]

    attention_mask = torch.tensor(
        sample["attention_mask"],
        dtype=torch.long,
        device="cpu",
    ).unsqueeze(0)              # [1, seq_len]

    return {
        "input_ids": input_ids,
        "attention_mask": attention_mask,
    }

# ============================================================
# 3. NVFP4 Recipe (GLM-4.5 MoE Safe Configuration)
# ============================================================

recipe = QuantizationModifier(
    targets="Linear",
    scheme="NVFP4",
    ignore=[
        "lm_head",              # Output head
        "re:.*mlp\\.gate$",     # MoE gate
    ],
)

# ============================================================
# 4. Perform NVFP4 Quantization
# ============================================================

print("Starting NVFP4 quantization (low VRAM mode)...")

oneshot(
    model=model,
    recipe=recipe,
    dataset=dataset,
    data_collator=data_collator,
    max_seq_length=MAX_SEQ_LEN,
    num_calibration_samples=NUM_CALIBRATION_SAMPLES,
    pipeline="sequential",           # Force layer-by-layer
    moe_calibrate_all_experts=True, # Only calibrate routed experts
)

print("Quantization finished.\n")

# ============================================================
# 5. Save compressed-tensors
# ============================================================

os.makedirs(SAVE_DIR, exist_ok=True)

model.save_pretrained(
    SAVE_DIR,
    save_compressed=True,
)

tokenizer.save_pretrained(SAVE_DIR)

print("==========================================")
print("NVFP4 Quantized model saved to:")
print(SAVE_DIR)
print("==========================================")
